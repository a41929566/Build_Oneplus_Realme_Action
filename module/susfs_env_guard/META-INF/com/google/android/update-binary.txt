#!/sbin/sh

#################
# Universal module installer bootstrap (Magisk / KernelSU / ReSukiSU / APatch)
#################

umask 022
ui_print() { echo "$1"; }

OUTFD=$2
ZIPFILE=$3

mount /data 2>/dev/null

# === 检测 Magisk + KernelSU 双 Root 环境，给出明确警告 ===
if [ "$KSU" = "true" ] && [ -f /data/adb/magisk/magiskd ]; then
    ui_print "!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!"
    ui_print "! 警告：检测到 Magisk + KernelSU 双 Root 环境！"
    ui_print "! 双 Root 会导致："
    ui_print "!   1. 属性伪装三通道不一致，检测工具易检出"
    ui_print "!   2. su 进程双重暴露，增加被检测面"
    ui_print "!   3. pkgmask/hwid_spoof 的 sysfs 节点可能竞态"
    ui_print "! 建议：保留其一运行——"
    ui_print "!   - 若使用 KSU 内核：先在 Magisk 中完全卸载"
    ui_print "!   - 若使用 Magisk 环境：请勿刷入 KSU 内核"
    ui_print "! 本模块将继续安装，但以上风险需知悉"
    ui_print "!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!"
    sleep 3
fi

# KernelSU / ReSukiSU / APatch 管理器使用自带安装流程。
# 兜底初始化：确认 spoof.conf 已生成（某些版本管理器可能不会自动调 customize.sh）
if [ "$KSU" = "true" ] || [ "$APATCH" = "true" ] || [ "$KSUD" = "true" ]; then
    CONF_DIR="/data/adb/susfs_env_guard"
    MOD_SRC="${MODPATH:-/data/adb/modules/susfs_env_guard}"
    if [ ! -f "$CONF_DIR/spoof.conf" ] && [ -f "$MOD_SRC/config/spoof.conf.example" ]; then
        mkdir -p "$CONF_DIR"
        cp "$MOD_SRC/config/spoof.conf.example" "$CONF_DIR/spoof.conf"
        ui_print "- spoof.conf 已初始化（KSU 兜底）"
    fi
    exit 0
fi

# 无 Magisk 运行库时（纯 KSU 环境偶发走到这里）也不要报错，交给管理器
if [ ! -f /data/adb/magisk/util_functions.sh ]; then
  exit 0
fi

# 标准 Magisk 流程
. /data/adb/magisk/util_functions.sh
[ "$MAGISK_VER_CODE" -lt 20400 ] && { ui_print "! Please use Magisk v20.4+"; exit 1; }
install_module
exit 0
